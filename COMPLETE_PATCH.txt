  function renderExpenses() {
    if (!workbook || !currentUser) {
      return null
    }

    const isAdmin = currentUser.Role === 'Admin'
    const selectedEmployeeId = isAdmin ? (activeEmployeeFilter || '') : currentUser.EmployeeID

    if (!isAdmin && !selectedEmployeeId) {
      return (
        <div className="tab-content">
          <p>Please select an employee to view expenses.</p>
        </div>
      )
    }

    return (
      <div className="tab-content">
        <ExpenseManager
          expenses={expenses}
          employeeId={selectedEmployeeId}
          isAdmin={isAdmin}
          currentUserId={currentUser.EmployeeID}
          onExpenseChange={saveExpenses}
        />
      </div>
